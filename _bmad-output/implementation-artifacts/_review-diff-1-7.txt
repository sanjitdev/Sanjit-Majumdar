"use client";
import type { ReactNode } from 'react';

/**
 * <NavCurrent> — AD-13 closed-set client component (8 of 10).
 *
 * Wraps a nav-link `<a>` and sets `aria-current="page"` when the active
 * route matches `href`. Both server-rendered (initial paint) and
 * client-rendered navigation paths carry the attribute, so screen-reader
 * users get the right announcement on every nav transition.
 *
 * Hover styling intentionally reuses `text-fg-2` (muted) → `text-fg`
 * (foreground) so the active link reads against the same color discipline
 * as the inactive ones — visual hierarchy is carried by `aria-current`,
 * not by color alone (AD-20 a11y floor).
 */
export function NavCurrent({
  href,
  isCurrent,
  children,
  className,
}: {
  href: string;
  isCurrent: boolean;
  children: ReactNode;
  className?: string;
}) {
  return (
    <a
      href={href}
      aria-current={isCurrent ? 'page' : undefined}
      className={[
        'text-fg-2 transition-colors hover:text-fg',
        className ?? '',
      ]
        .filter(Boolean)
        .join(' ')}
    >
      {children}
    </a>
  );
}

"use client";
import { useEffect } from 'react';

/**
 * <ScrollProgress> — AD-13 closed-set client component (2 of 10).
 *
 * Renders a 1px-tall `--accent` bar pinned to the viewport bottom edge.
 * `transform: scaleX(var(--progress))` is driven by a passive scroll listener
 * that writes `--progress` (0–1) to `:root`.
 *
 * Motion gate: under `prefers-reduced-motion: reduce` the listener is a no-op
 * and `--progress` stays at 0; the bar remains visible but pinned at 0%
 * scaleX (final-state guarantee per AD-16 + AD-20).
 *
 * The bar is `position: fixed` at viewport bottom (NOT inside the header) so
 * it tracks scroll position across the WHOLE page, not just within the nav
 * (UX-DR4 contract).
 */
export function ScrollProgress() {
  useEffect(() => {
    if (typeof window === 'undefined') return;

    const root = document.documentElement;
    const reducedMql = window.matchMedia('(prefers-reduced-motion: reduce)');

    const update = () => {
      if (reducedMql.matches) return;
      const scrollY = window.scrollY || window.pageYOffset || 0;
      const docHeight =
        document.documentElement.scrollHeight - window.innerHeight;
      const progress = docHeight > 0 ? scrollY / docHeight : 0;
      const clamped = Math.max(0, Math.min(1, progress));
      root.style.setProperty('--progress', String(clamped));
    };

    if (reducedMql.matches) {
      root.style.setProperty('--progress', '0');
    } else {
      update();
    }

    window.addEventListener('scroll', update, { passive: true });
    return () => {
      window.removeEventListener('scroll', update);
    };
  }, []);

  return (
    <div
      aria-hidden="true"
      className="pointer-events-none fixed bottom-0 left-0 right-0 z-40 h-px w-full origin-left bg-[var(--accent)] transition-transform duration-100"
      style={{ transform: 'scaleX(var(--progress, 0))' }}
    />
  );
}
